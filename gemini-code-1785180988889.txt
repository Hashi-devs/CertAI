            const file = e.target.files[0];
            if (!file) return;

            const fontName = 'CustomFont_' + Math.random().toString(36).substr(2, 6);
            
            try {
                const buffer = await file.arrayBuffer();
                const customFont = new FontFace(fontName, buffer);
                await customFont.load();
                document.fonts.add(customFont);

                const select = document.getElementById('font-family-select');
                const opt = document.createElement('option');
                opt.value = fontName;
                opt.innerText = `Uploaded: ${file.name}`;
                opt.selected = true;
                select.prepend(opt);

                updateTextStyle('fontFamily', fontName);
                showToast("Font Uploaded Successfully", `Custom font "${file.name}" ready to use.`);
            } catch (err) {
                console.error('Font load error:', err);
                alert('Could not load font file. Please ensure it is a valid .ttf or .otf file.');
            }
        }

        function handleCSVUpload(e) {
            const file = e.target.files[0];
            if (!file) return;

            const reader = new FileReader();
            reader.onload = (event) => {
                const text = event.target.result;
                const lines = text.split(/\r?\n/).map(l => l.trim()).filter(l => l.length > 0);

        // ULTRA HD SUPER-SAMPLED RENDER ENGINE
        function renderCanvas(overrideText = null) {
            if (!templateImage.width) return;

            // Get original template native dimensions
            let nativeW = templateImage.naturalWidth || templateImage.width;
            let nativeH = templateImage.naturalHeight || templateImage.height;

            // Ensure super-sampling multiplier so text rendering is ultra-crisp (minimum 2400px width)
            const hdMultiplier = Math.max(1, 2400 / nativeW);
            const masterW = Math.round(nativeW * hdMultiplier);
            const masterH = Math.round(nativeH * hdMultiplier);

            canvas.width = masterW;
            canvas.height = masterH;

            // High-fidelity anti-aliasing
            ctx.imageSmoothingEnabled = true;
            ctx.imageSmoothingQuality = 'high';

            // Background Fill Pass (Prevents black canvas when saving JPEG format)
            if (exportFormat === 'jpeg') {
                ctx.fillStyle = '#ffffff';
                ctx.fillRect(0, 0, masterW, masterH);
            }

            // Draw template design scaled up losslessly
            ctx.drawImage(templateImage, 0, 0, masterW, masterH);

            const activeName = overrideText || names[currentIdx] || "Alexander Bartholomew";
            const x = (style.xPct / 100) * masterW;
            const y = (style.yPct / 100) * masterH;

            // Scale font size according to the HD master canvas resolution
            const scaleFactor = masterW / 1200;
            let computedFontSize = style.fontSize * scaleFactor;

            if (autoFitEnabled) {
                const maxAvailableWidth = masterW * 0.82;
                ctx.font = `${style.fontWeight} ${computedFontSize}px "${style.fontFamily}", sans-serif`;
                let metrics = ctx.measureText(activeName);

                while (metrics.width > maxAvailableWidth && computedFontSize > 16) {
                    computedFontSize -= 2;
                    ctx.font = `${style.fontWeight} ${computedFontSize}px "${style.fontFamily}", sans-serif`;
                    metrics = ctx.measureText(activeName);
                }
            }

            ctx.save();
            ctx.font = `${style.fontWeight} ${computedFontSize}px "${style.fontFamily}", sans-serif`;
            ctx.fillStyle = style.color;
            ctx.textAlign = style.align;
            ctx.textBaseline = 'middle';

            ctx.fillText(activeName, x, y);

            // Dashed slotting boundary box during editing mode
            if (!overrideText) {
                const metrics = ctx.measureText(activeName);
                const w = metrics.width + (30 * scaleFactor);
                const h = computedFontSize + (15 * scaleFactor);

                let startX = x - w / 2;
                if (style.align === 'left') startX = x - (15 * scaleFactor);
                if (style.align === 'right') startX = x - w + (15 * scaleFactor);

                ctx.strokeStyle = '#2563eb';
                ctx.lineWidth = Math.max(2, 3 * scaleFactor);
                ctx.setLineDash([8 * scaleFactor, 6 * scaleFactor]);
                ctx.strokeRect(startX, y - h / 2, w, h);
            }

            ctx.restore();

            document.getElementById('candidate-counter').innerText = `${currentIdx + 1} / ${names.length}`;
        }

        function updateTextStyle(param, val) {
            style[param] = val;

            if (param === 'fontSize') {
                document.getElementById('size-input').value = val;
                document.getElementById('size-slider').value = val;
            }
            if (param === 'color') {
                document.getElementById('color-picker').value = val;
                document.getElementById('color-hex-text').value = val;
            }
            if (param === 'xPct') document.getElementById('x-indicator').innerText = `${Math.round(val)}%`;
            if (param === 'yPct') document.getElementById('y-indicator').innerText = `${Math.round(val)}%`;
            if (param === 'align') updateAlignUI(val);

            renderCanvas();
        }

        function updateAlignUI(activeAlign) {
            ['left', 'center', 'right'].forEach(a => {
                const btn = document.getElementById(`align-${a}-btn`);
                if (btn) {
                    if (a === activeAlign) {
                        btn.className = "flex-1 py-1 text-center text-xs bg-black dark:bg-white text-white dark:text-black rounded font-bold transition shadow-sm";
                    } else {
                        btn.className = "flex-1 py-1 text-center text-xs text-neutral-500 hover:text-black dark:hover:text-white hover:bg-neutral-100 dark:hover:bg-neutral-800 rounded transition";
                    }
                }
            });
        }

        function setupDragAndDrop() {
            const handlePointer = (e) => {
                if (positionLocked) return;

                const rect = canvas.getBoundingClientRect();
                const clientX = e.touches ? e.touches[0].clientX : e.clientX;
                const clientY = e.touches ? e.touches[0].clientY : e.clientY;

                const x = ((clientX - rect.left) / rect.width) * 100;
                const y = ((clientY - rect.top) / rect.height) * 100;

                if (x >= 0 && x <= 100 && y >= 0 && y <= 100) {
                    style.xPct = x;
                    style.yPct = y;

                    document.getElementById('x-slider').value = Math.round(x);
                    document.getElementById('y-slider').value = Math.round(y);
                    document.getElementById('x-indicator').innerText = `${Math.round(x)}%`;
                    document.getElementById('y-indicator').innerText = `${Math.round(y)}%`;

                    renderCanvas();
                }
            };

            canvas.addEventListener('mousedown', (e) => { isDragging = true; handlePointer(e); });
            window.addEventListener('mousemove', (e) => { if (isDragging) handlePointer(e); });
            window.addEventListener('mouseup', () => { isDragging = false; });

            canvas.addEventListener('touchstart', (e) => { isDragging = true; handlePointer(e); });
            window.addEventListener('touchmove', (e) => { if (isDragging) handlePointer(e); });
            window.addEventListener('touchend', () => { isDragging = false; });
        }

        function getMimeType() {
            if (exportFormat === 'jpeg') return 'image/jpeg';
            if (exportFormat === 'webp') return 'image/webp';
            return 'image/png';
        }

        function downloadSinglePNG() {
            const name = names[currentIdx] || 'Sample';
            const ext = exportFormat;
            
            // Clean render pass without dashed edit boundary
            renderCanvas(name);

            const mime = getMimeType();
            
            canvas.toBlob((blob) => {
                const url = URL.createObjectURL(blob);
                const link = document.createElement('a');
                link.download = `Certificate_${name.replace(/\s+/g, '_')}.${ext}`;
                link.href = url;
                document.body.appendChild(link);
                link.click();
                document.body.removeChild(link);
                setTimeout(() => URL.revokeObjectURL(url), 4000);
                
                renderCanvas(); // Restore live editor bounding box
            }, mime, 1.0); // Maximum 100% Quality Ratio
        }

        // ULTRA-FAST & ULTRA-HD LOSSLESS ZIP EXPORTER
        async function exportZipArchive() {
            if (names.length === 0) return;

            const modal = document.getElementById('progress-modal');
            const status = document.getElementById('progress-text');
            const bar = document.getElementById('progress-fill');

            modal.classList.remove('hidden');

            const zip = new JSZip();
            const folder = zip.folder("Certificates");
            const ext = exportFormat;
            const mime = getMimeType();

            for (let i = 0; i < names.length; i++) {
                const name = names[i];
                status.innerText = `Rendering Ultra-HD ${i + 1} of ${names.length}: ${name}`;
                bar.style.width = `${Math.round(((i + 1) / names.length) * 100)}%`;

                await new Promise(r => setTimeout(r, 10));

                // Clean HD Render pass
                renderCanvas(name);

                // Full 1.0 Lossless Blob Creation
                const blob = await new Promise(resolve => canvas.toBlob(resolve, mime, 1.0));
                folder.file(`Certificate_${name.replace(/[^a-zA-Z0-9]/g, "_")}.${ext}`, blob);
            }

            status.innerText = "Packing ZIP file...";
            bar.style.width = "100%";

            const zipContent = await zip.generateAsync({
                type: "blob",
                compression: "STORE"
            }, function updateCallback(metadata) {
                status.innerText = `Packing: ${Math.round(metadata.percent)}%`;
            });

            const zipUrl = URL.createObjectURL(zipContent);
            const downloadLink = document.createElement('a');
            downloadLink.href = zipUrl;
            downloadLink.download = `CertAI_Certificates.zip`;
            document.body.appendChild(downloadLink);
            downloadLink.click();
            document.body.removeChild(downloadLink);

            setTimeout(() => URL.revokeObjectURL(zipUrl), 5000);

            modal.classList.add('hidden');
            renderCanvas();
            showToast("ZIP Downloaded!", "High-definition certificate batch downloaded.");
        }
